from pathlib import Path

r=Path('/tmp/airs-node-judge-20260920/release');spec=json.loads((r/'SPEC.json').read_text());staging=json.loads((r/'STAGING.json').read_text())
# Staging completion and staged immutable package metadata are prerequisites.
if staging.get('exit_code',0)!=0 or not (r/'staged/NPM-PACKAGES.json').is_file():raise RuntimeError('Staging incomplete')
selected=[r/'SPEC.json']
selected+=sorted(p for p in (r/'staged').rglob('*') if p.is_file())
for row in spec['platforms']:

 if not (d/'ACCEPTANCE.json').is_file():raise RuntimeError('Missing native candidate evidence')
