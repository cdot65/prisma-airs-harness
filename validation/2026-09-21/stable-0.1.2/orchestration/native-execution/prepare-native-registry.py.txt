from pathlib import Path
import hashlib
import json
import plistlib
import shutil
import subprocess
import tarfile
import uuid

cache = Path('/Users/cdot/.cache/airs-012-20260921')
venv = Path('/Users/cdot/.cache/airs-native-mcp-20260919/venv')
root = cache / 'stable012-candidate'
archive = cache / 'registry-inputs.tar'
manifest = json.loads((cache / 'REGISTRY-TRANSFER.json').read_text())
with archive.open('rb') as stream:
    actual = hashlib.file_digest(stream, 'sha256').hexdigest()
if actual != __import__('sys').argv[1] or actual != manifest['archive_sha256']:
    raise RuntimeError('Registry transfer hash mismatch')
source = cache / 'registry-inputs-stable012'
source.mkdir(mode=0o700)
with tarfile.open(archive) as tar:
    members = tar.getmembers()
    if any(not member.isfile() for member in members) or len(members) != len(manifest['files']) or {member.name for member in members} != set(manifest['files']):
        raise RuntimeError('Unexpected registry input archive inventory')
    tar.extractall(source, filter='data')
for relative, expected in manifest['files'].items():
    src = source / relative
    with src.open('rb') as stream:
        if hashlib.file_digest(stream, 'sha256').hexdigest() != expected:
            raise RuntimeError('Registry input file changed')
    dest = root / relative
    if dest.exists():
        with dest.open('rb') as stream:
            if hashlib.file_digest(stream, 'sha256').hexdigest() != expected:
                raise RuntimeError('Refusing different existing input: ' + relative)
    else:
        dest.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(src, dest)
shutil.copy2(cache / 'REGISTRY-TRANSFER.json', source / 'TRANSFER.json')
if json.loads((root / 'SPEC.json').read_text()) != manifest['spec']:
    raise RuntimeError('Registry SPEC mismatch')

r=subprocess.run(command)
