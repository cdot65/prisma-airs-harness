from pathlib import Path
import json,zipfile,hashlib
root=Path('/tmp/airs-upstream-r2-20260922/release');dest=root/'mac-signing-evidence'
source=(root/'SOURCE').read_text().strip();native=json.loads((root/'mac-native/EXTRACTION.json').read_text())
with zipfile.ZipFile(root/'mac-signed-candidate.zip') as z:
 names=['SIGNING.json','NOTARIZATION.json','NATIVE-STORE.json','NATIVE-BUILD.json','NATIVE-KEYCHAIN.json','native-tests.log']
 signing=json.loads(z.read('evidence/SIGNING.json'))
 assert signing['source_commit']==source and signing['binary_sha256']==native['binary_sha256']
 assert signing['codesign_verified'] and signing['notarization_verified'] and signing['hardened_runtime']
 assert json.loads(z.read('evidence/NOTARIZATION.json'))['status']=='Accepted'
 assert json.loads(z.read('evidence/NATIVE-STORE.json'))['passed']
 assert json.loads(z.read('evidence/NATIVE-KEYCHAIN.json'))['passed']
 names+=list(signing['verification_log_sha256'])
 dest.mkdir(exist_ok=True)
 for name in names:
  assert Path(name).name==name
  data=z.read('evidence/'+name);assert len(data)<20*1024*1024
  if name in signing['verification_log_sha256']:assert hashlib.sha256(data).hexdigest()==signing['verification_log_sha256'][name]
  p=dest/name
  if p.exists():assert p.read_bytes()==data
  else:p.write_bytes(data)
 print(json.dumps({'passed':True,'files':len(names),'signed_binary_sha256':native['binary_sha256']}))
