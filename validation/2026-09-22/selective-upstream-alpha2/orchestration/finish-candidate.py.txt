from pathlib import Path
import json, shutil, subprocess, sys, time
r=Path(__file__).resolve().parent
end=time.monotonic()+8*3600
local=r/'upstream013r2-candidate'
remote=r/'remote-candidate-evidence'
while not all(p.exists() for p in [local/'HOST-COMPLETE.json',remote/'SUMMARY.json',r/'WORKSPACE-REVIEW.json']):
 if (local/'HOST-COMPLETE.json').exists():
  assert json.loads((local/'HOST-COMPLETE.json').read_text())['passed'], 'Local acceptance failed'
 if time.monotonic()>end:raise TimeoutError('Candidate or workspace evidence still pending')
 time.sleep(30)
assert json.loads((local/'HOST-COMPLETE.json').read_text())['passed']
workspace=json.loads((r/'WORKSPACE-REVIEW.json').read_text())
assert workspace['only_reviewed_baseline_failures'] and workspace['failed']==3
spec=json.loads((r/'SPEC.json').read_text())
assert workspace['source_commit']==spec['source_commit']
for row in spec['platforms']:
 target=row['target']; source=local if target.startswith('x86_64') else remote
 shutil.copytree(source/'candidate-acceptance'/target,r/'candidate-acceptance'/target)
 for phase in ['accept-dispatch','extras-dispatch']:
  report=json.loads((source/phase/target/'COMPLETE.json').read_text())
  assert report['passed'] and report['expected_native_sha256']==row['binary_sha256']
  assert all(c['exit_code']==0 for c in report['commands'])
  if phase=='extras-dispatch':
   assert {'typesafe-node','test_airs_policy_retry','test_airs_doctor'} <= {c['name'] for c in report['commands']}
  shutil.copytree(source/phase/target,r/'native-extra-evidence'/phase/target)
base=[sys.executable,str(r/'release-frozen-upstream013r2/scripts/airs_test_release.py')]
for name,command in [
 ('CANDIDATE-VERIFICATION',base+['verify-all','--spec',str(r/'SPEC.json'),'--acceptance',str(r/'candidate-acceptance')]),
 ('STAGING',base+['stage','--spec',str(r/'SPEC.json'),'--packages',str(r/'candidate-npm'),'--acceptance',str(r/'candidate-acceptance'),'--output',str(r/'staged')])]:
 with (r/(name+'.log')).open('x') as log:p=subprocess.run(command,stdout=log,stderr=subprocess.STDOUT)
 (r/(name+'.json')).write_text(json.dumps({'exit_code':p.returncode,'source_commit':spec['source_commit'],'command':command},indent=2)+'\n')
 assert p.returncode==0, name+' failed'
print('All three native candidates verified and staged; publication not performed.')
