from pathlib import Path
import hashlib
import json
import plistlib
import shutil
import subprocess
import tarfile
import uuid

cache = Path('/Users/cdot/.cache/airs-routing-20260925-r2')
venv = Path('/Users/cdot/.cache/airs-native-mcp-20260919/venv')
import sys
expected_archive = sys.argv[1]
archive = cache / 'candidate-transfer.tar'
transfer = json.loads((cache / 'CANDIDATE-TRANSFER.json').read_text())
with archive.open('rb') as stream:
    actual = hashlib.file_digest(stream, 'sha256').hexdigest()
if actual != expected_archive or actual != transfer['archive_sha256']:
    raise RuntimeError('Candidate transfer hash mismatch')
root = cache / 'routing013r5-candidate'
root.mkdir(mode=0o700)
packages = root / 'candidate-npm'
packages.mkdir(mode=0o700)
with tarfile.open(archive) as tar:
    members = tar.getmembers()
    if any(not member.isfile() for member in members) or len(members) != len(transfer['files']) or {member.name for member in members} != set(transfer['files']):
        raise RuntimeError('Unexpected candidate archive members')
    tar.extractall(packages, filter='data')
for relative, expected in transfer['files'].items():
    with (packages / relative).open('rb') as stream:
        if hashlib.file_digest(stream, 'sha256').hexdigest() != expected:
            raise RuntimeError('Extracted candidate file mismatch')
shutil.copy2(cache / 'SPEC.json', root / 'SPEC.json')
if json.loads((root / 'SPEC.json').read_text()) != transfer['spec']:
    raise RuntimeError('Specification mismatch')
shutil.copy2(cache / 'CANDIDATE-TRANSFER.json', root / 'TRANSFER.json')

  if r.returncode:raise SystemExit(r.returncode)
finally:(root/'HOST-COMPLETE.json').write_text(json.dumps({'commands':commands,'passed':len(commands)==2 and all(r['exit_code']==0 for r in commands)})+'\\n')
'''
(root / 'host-runner.py').write_text(runner)
tmp = root / 'tmp'
tmp.mkdir(mode=0o700)
label = 'io.cdot.airs.typesafe.20260920.candidate.' + uuid.uuid4().hex
plist = root / 'mac-job.plist'
plist.write_bytes(plistlib.dumps({
    'Label': label, 'ProgramArguments': [str(venv / 'bin/python'), str(root / 'host-runner.py')],
    'RunAtLoad': True, 'WorkingDirectory': str(root),
    'EnvironmentVariables': {'PATH': str(venv / 'bin') + ':/opt/homebrew/opt/openssl@3/bin:/opt/homebrew/bin:/usr/local/bin:/usr/bin:/bin',

subprocess.run(['launchctl', 'bootstrap', 'gui/501', str(plist)], check=True)
report = {'mac_gui_label': label, 'mac_files_verified': len(transfer['files']), 'archive_sha256': actual, 'spec': transfer['spec'], 'mac_root': str(root), 'production_acceptance': False}
