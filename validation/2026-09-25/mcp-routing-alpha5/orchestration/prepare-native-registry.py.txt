from pathlib import Path
import hashlib
import json
import plistlib
import shutil
import subprocess
import tarfile
import uuid

cache = Path('/Users/cdot/.cache/airs-routing-20260925-r2')
venv = Path('/Users/cdot/.cache/airs-native-mcp-20260919/venv')
root = cache / 'routing013r5-candidate'
archive = cache / 'registry-inputs.tar'
manifest = json.loads((cache / 'REGISTRY-TRANSFER.json').read_text())
with archive.open('rb') as stream:
    actual = hashlib.file_digest(stream, 'sha256').hexdigest()
if actual != __import__('sys').argv[1] or actual != manifest['archive_sha256']:
    raise RuntimeError('Registry transfer hash mismatch')
source = cache / 'registry-inputs-selection013r4'
source.mkdir(mode=0o700)
with tarfile.open(archive) as tar:
    members = tar.getmembers()
    if any(not member.isfile() for member in members) or len(members) != len(manifest['files']) or {member.name for member in members} != set(manifest['files']):
        raise RuntimeError('Unexpected registry input archive inventory')
    tar.extractall(source, filter='data')
for relative, expected in manifest['files'].items():
    src = source / relative
    with src.open('rb') as stream:
        if hashlib.file_digest(stream, 'sha256').hexdigest() != expected:
            raise RuntimeError('Registry input file changed')
    dest = root / relative
    if dest.exists():
        with dest.open('rb') as stream:
            if hashlib.file_digest(stream, 'sha256').hexdigest() != expected:
                raise RuntimeError('Refusing different existing input: ' + relative)
    else:
        dest.parent.mkdir(parents=True, exist_ok=True)
        shutil.copy2(src, dest)
shutil.copy2(cache / 'REGISTRY-TRANSFER.json', source / 'TRANSFER.json')
if json.loads((root / 'SPEC.json').read_text()) != manifest['spec']:
    raise RuntimeError('Registry SPEC mismatch')
runner = '''from pathlib import Path
import subprocess,json,sys
root=Path(__file__).resolve().parent
command=[sys.executable,str(root.parent/'native-worker.py'),'verify','--root',str(root),'--tooling',str(root.parent/'frozen-tooling'),'--packages',str(root/'staged')]
r=subprocess.run(command)
(root/'REGISTRY-HOST-COMPLETE.json').write_text(json.dumps({'exit_code':r.returncode,'passed':r.returncode==0,'command':command})+'\\n')
raise SystemExit(r.returncode)
'''
(root / 'registry-runner.py').write_text(runner)
tmp = root / 'registry-tmp'
tmp.mkdir(mode=0o700)
label = 'io.cdot.airs.typesafe.20260920.registry.' + uuid.uuid4().hex
plist = root / 'registry-job.plist'
plist.write_bytes(plistlib.dumps({
    'Label': label, 'ProgramArguments': [str(venv / 'bin/python'), str(root / 'registry-runner.py')],
    'RunAtLoad': True, 'WorkingDirectory': str(root),
    'EnvironmentVariables': {'PATH': str(venv / 'bin') + ':/opt/homebrew/opt/openssl@3/bin:/opt/homebrew/bin:/usr/local/bin:/usr/bin:/bin',
                             'HOME': '/Users/cdot', 'TERM': 'xterm-256color', 'TMPDIR': str(tmp), 'PYTHONDONTWRITEBYTECODE': '1'},
    'StandardOutPath': str(root / 'MAC-REGISTRY.log'), 'StandardErrorPath': str(root / 'MAC-REGISTRY.log'),
}))
subprocess.run(['launchctl', 'bootstrap', 'gui/501', str(plist)], check=True)
report = {'mac_gui_label': label, 'mac_files_verified': len(manifest['files']), 'archive_sha256': actual, 'installation': 'registry', 'spec': manifest['spec'], 'production_acceptance': False}
(root / 'REGISTRY-LAUNCH.json').write_text(json.dumps(report, indent=2) + '\n')
print(json.dumps(report))
