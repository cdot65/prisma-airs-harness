fn main() {
 let path = "/tmp/airs-musl-statx-fixture";
 std::fs::write(path, "fixture").unwrap();
 let file=std::fs::File::open("/tmp").unwrap();
 eprintln!("std birthtime available: {}", std::fs::metadata(path).unwrap().created().is_ok());
 let result=rustix::fs::statx(&file, "airs-musl-statx-fixture", rustix::fs::AtFlags::SYMLINK_NOFOLLOW, rustix::fs::StatxFlags::BASIC_STATS | rustix::fs::StatxFlags::BTIME);
 match result { Ok(value) => eprintln!("rustix birthtime mask: {}", value.stx_mask & rustix::fs::StatxFlags::BTIME.bits()), Err(error) => eprintln!("rustix statx error: {error:?}") };
 std::fs::remove_file(path).unwrap();
}
