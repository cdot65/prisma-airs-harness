#!/opt/homebrew/bin/python3.13
"""Give each native test its own disposable tmp root; retain failures for review."""
import errno,json,os,shutil,subprocess,sys,tempfile,time
from pathlib import Path
root=Path('/Users/cdot/.cache/airs-upstream-20260926')
owned=Path(tempfile.mkdtemp(prefix='airs-t-',dir='/tmp')).resolve()
env=dict(os.environ,TMPDIR=str(owned)+'/',TMP=str(owned),TEMP=str(owned))
result=subprocess.run([sys.executable,str(root/'source/scripts/airs_rust_test_runner.py'),*sys.argv[1:]],env=env)
if result.returncode==0:
 # Node helpers may finish writing their compile cache just after the test exits.
 # Retry only a concurrent nonempty-directory race; preserve other cleanup failures.
 for attempt in range(11):
  try:
   shutil.rmtree(owned)
   break
  except FileNotFoundError:
   break
  except OSError as exc:
   if exc.errno != errno.ENOTEMPTY or attempt == 10:
    raise
   time.sleep(0.2)
else:
 with (root/'unicode-native-failed-temp.jsonl').open('a') as f:f.write(json.dumps({'test_args':sys.argv[1:],'tmp':str(owned),'exit':result.returncode})+'\n')
raise SystemExit(result.returncode)
